echo "Enumerating Subdomains of $1 : STARTED"

echo "Number of subdomains : $(wc -l < subdomains)"
echo "Resolving..."
cat subdomains | dnsx -stats -v |anew live-subdomains >/dev/null
echo "Number of resolved subdomains : $(wc -l < live-subdomains)"
echo "Bruteforcing..."
